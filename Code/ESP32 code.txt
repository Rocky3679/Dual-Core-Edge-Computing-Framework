 *             True Instantaneous PF (P/S integration), Reactive Power (VAR),
 *             2-Stage Capacitor Bank APFC, and Automated Load Shedding (Motor 2).

#include <Wire.h>
#include <WiFi.h>

// ── 1. NETWORK & FIREBASE CONFIGURATION ──────────────────────────────────────────
#define WIFI_SSID           "YOUR_WIFI_SSID"
#define WIFI_PASSWORD       "YOUR_WIFI_PASSWORD"
#define FIREBASE_HOST       "unitysync-4975d-default-rtdb.asia-southeast1.firebasedatabase.app"
#define FIREBASE_AUTH       "YOUR_FIREBASE_SECRET_HERE"

// ── 2. GPIO PIN DEFINITIONS ──────────────────────────────────────────────────────
// Analog Inputs (ADC1 only — safe with WiFi active)
#define PIN_VSENSOR         34   // ZMPT101B AC Voltage Sensor
#define PIN_CS1             35   // ACS712 AC Mains Current (230V rail)
#define PIN_CS2             32   // ACS712 Total Load Current (Motor 1 + Motor 2)
#define PIN_CS3             33   // ACS712 Generation Current (Solar + Wind)

// Relay Outputs (Active-LOW: LOW = ON / Closed, HIGH = OFF / Open)
#define PIN_RELAY_CAP1      25   // Capacitor Bank 1 (10 uF APFC Stage 1)
#define PIN_RELAY_CAP2      26   // Capacitor Bank 2 (10 uF APFC Stage 2)
#define PIN_RELAY_SOLAR     27   // Solar Source Inverter/Bus Switch
#define PIN_RELAY_WIND      14   // Wind Source Inverter/Bus Switch
#define PIN_RELAY_MOTOR1    12   // Motor 1 (Critical Load — Always Retained)
#define PIN_RELAY_MOTOR2    13   // Motor 2 (Non-Critical Load — Sheddable)

#define RELAY_ON            LOW
#define RELAY_OFF           HIGH

// ── 3. SENSOR CALIBRATION & THRESHOLDS ───────────────────────────────────────────
#define ADC_VREF            3.3f
#define ADC_RES             4095.0f
#define ADC_MIDPOINT        2048         // Nominal 1.65V zero-crossing for 3.3V ADC
#define SAMPLE_COUNT        400          // 400 samples @ 100us = 40ms (Exactly two 50Hz cycles)
#define SAMPLE_INTERVAL_US  100          // 10 kHz sampling rate

#define ACS_SENSITIVITY     0.100f       // V/A (ACS712-20A = 0.100, 5A = 0.185, 30A = 0.066)
#define ZMPT_SCALE          191.7f       // Calibrate: 230.0V / raw_rms_secondary_volts
#define CURRENT_NOISE_FLOOR 0.04f        // Cutoff to prevent ghost readings

// Decision Thresholds
#define AC_MIN_VRMS         50.0f        // Below this = AC Grid Failure
#define GEN_MIN_CURRENT     0.15f        // Below this = Generation deficit (Triggers Shedding)
#define PF_THRESH_BOTH_ON   0.90f        // PF < 0.90 -> Turn ON Cap1 + Cap2
#define PF_THRESH_ONE_ON    0.95f        // PF < 0.95 -> Turn ON Cap1 only
#define PF_THRESH_ALL_OFF   0.98f        // PF > 0.98 -> Turn OFF all caps (Avoid leading PF)
#define TARIFF_PER_KWH      8.50f        // Electricity board tariff (Rs./kWh)

// ── 4. THREAD-SAFE SHARED STATE (FreeRTOS Mutex Protected) ──────────────────────
struct SystemTelemetry {
  float vrms;
  float imains;
  float iload;
  float igen;
  float realPower;       // P (Watts)
  float apparentPower;   // S (VA)
  float reactivePower;   // Q (VAR)
  float powerFactor;     // True Instantaneous PF (P/S)
  float energyConsumed;  // Total kWh
  float energySaved;     // Renewable kWh
  float billAmount;      // Cumulative Rs.
  bool gridAvailable;
  bool solarActive;
  bool windActive;
  bool cap1Active;
  bool cap2Active;
  bool motor1Active;
  bool motor2Active;     // False when load is shed
  bool loadShedActive;
};

volatile SystemTelemetry sharedData;
SemaphoreHandle_t telemetryMutex;

// LCD & Cloud Objects
LiquidCrystal_I2C *lcd = nullptr;
FirebaseData fbData;
FirebaseAuth fbAuth;
FirebaseConfig fbConfig;

// Task Prototypes
void controlCore0Task(void *pvParameters);
void displayCloudCore1Task(void *pvParameters);
uint8_t scanI2CAddress();
void setRelay(int pin, bool state);
String getUptimeString();

// ================================================================================
// SETUP INITIALIZATION (Runs on Core 1 at Boot)
// ================================================================================
void setup() {
  Serial.begin(115200);
  Serial.println("\n[BOOT] Initializing UnitySync Dual-Core Framework...");

  // 1. Hardware Pin Initialization (Enforce ALL Relays Safe OFF immediately)
  const int relayPins[] = {PIN_RELAY_CAP1, PIN_RELAY_CAP2, PIN_RELAY_SOLAR, 
                           PIN_RELAY_WIND, PIN_RELAY_MOTOR1, PIN_RELAY_MOTOR2};
  for (int pin : relayPins) {
    pinMode(pin, OUTPUT);
    digitalWrite(pin, RELAY_OFF);
  }

  // 2. ADC Hardware Configuration
  analogReadResolution(12);
  analogSetAttenuation(ADC_11db);

  // 3. Thread Synchronization Mutex
  telemetryMutex = xSemaphoreCreateMutex();

  // 4. Auto-detect I2C Address and Initialize LCD
  Wire.begin(21, 22);
  uint8_t detectedI2C = scanI2CAddress();
  lcd = new LiquidCrystal_I2C(detectedI2C, 16, 2);
  lcd->init();
  lcd->backlight();
  lcd->clear();
  lcd->setCursor(0, 0); lcd->print("UnitySync EdgeFW");
  lcd->setCursor(0, 1); lcd->print("Starting Cores..");
  delay(1200);

  // 5. Initialize WiFi & Firebase
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 15) {
    delay(400);
    Serial.print(".");
    attempts++;
  }
  
  fbConfig.host = FIREBASE_HOST;
  fbConfig.signer.tokens.legacy_token = FIREBASE_AUTH;
  Firebase.begin(&fbConfig, &fbAuth);
  Firebase.reconnectWiFi(true);

  // 6. Launch Tasks onto Dedicated Hardware Cores
  // Core 0: High-Priority Real-Time Sensing & Power Control (Priority 2)
  xTaskCreatePinnedToCore(controlCore0Task, "ControlCore0", 8192, NULL, 2, NULL, 0);

  // Core 1: Background LCD UI & Firebase RTDB Sync (Priority 1)
  xTaskCreatePinnedToCore(displayCloudCore1Task, "DisplayCloudCore1", 8192, NULL, 1, NULL, 1);

  Serial.println("[BOOT] Dual Cores active and synchronized.");
}

void loop() {
  // Empty — FreeRTOS scheduler handles all execution on both cores
  vTaskDelay(portMAX_DELAY);
}

// ================================================================================
// CORE 0: SYNCHRONIZED SENSING, TRUE PF CALCULATION & APFC/LOAD LOGIC
// ================================================================================
void controlCore0Task(void *pvParameters) {
  unsigned long lastEnergyTimestamp = millis();
  float localEkWh = 0.0f;
  float localESaved = 0.0f;

  for (;;) {
    // ── STEP 1: Synchronized Dual-Channel Sampling for Instantaneous Real Power
    double sumInstantaneousPower = 0.0;
    double sumVSquared = 0.0;
    double sumISquaredMains = 0.0;
    double sumISquaredLoad = 0.0;
    double sumISquaredGen = 0.0;

    for (int i = 0; i < SAMPLE_COUNT; i++) {
      long rawV  = analogRead(PIN_VSENSOR) - ADC_MIDPOINT;
      long rawI1 = analogRead(PIN_CS1)     - ADC_MIDPOINT;
      long rawI2 = analogRead(PIN_CS2)     - ADC_MIDPOINT;
      long rawI3 = analogRead(PIN_CS3)     - ADC_MIDPOINT;

      sumInstantaneousPower += (double)(rawV * rawI1);
      sumVSquared           += (double)(rawV * rawV);
      sumISquaredMains      += (double)(rawI1 * rawI1);
      sumISquaredLoad       += (double)(rawI2 * rawI2);
      sumISquaredGen        += (double)(rawI3 * rawI3);

      delayMicroseconds(SAMPLE_INTERVAL_US);
    }

    // ── STEP 2: Mathematical Engineering Derivations
    // Voltage & Current RMS
    float vRms = ((sqrt(sumVSquared / SAMPLE_COUNT) / ADC_RES) * ADC_VREF) * ZMPT_SCALE;
    if (vRms < 10.0f) vRms = 0.0f;

    float iMains = (((sqrt(sumISquaredMains / SAMPLE_COUNT) / ADC_RES) * ADC_VREF) / ACS_SENSITIVITY);
    if (iMains < CURRENT_NOISE_FLOOR) iMains = 0.0f;

    float iLoad = (((sqrt(sumISquaredLoad / SAMPLE_COUNT) / ADC_RES) * ADC_VREF) / ACS_SENSITIVITY);
    if (iLoad < CURRENT_NOISE_FLOOR) iLoad = 0.0f;

    float iGen = (((sqrt(sumISquaredGen / SAMPLE_COUNT) / ADC_RES) * ADC_VREF) / ACS_SENSITIVITY);
    if (iGen < CURRENT_NOISE_FLOOR) iGen = 0.0f;

    // True Instantaneous Power Factor Calculation: PF = P / S
    float apparentPower = vRms * iMains; // S = Vrms * Irms (VA)
    float realPower = 0.0f;
    float reactivePower = 0.0f;
    float powerFactor = 1.0f;

    if (vRms >= AC_MIN_VRMS && iMains > CURRENT_NOISE_FLOOR) {
      double rawAvgP = sumInstantaneousPower / SAMPLE_COUNT;
      realPower = abs((float)(((rawAvgP / (ADC_RES * ADC_RES)) * (ADC_VREF * ADC_VREF) * ZMPT_SCALE) / ACS_SENSITIVITY));
      
      if (apparentPower > 1.0f) {
        powerFactor = constrain(realPower / apparentPower, 0.0f, 1.0f);
      }
      
      float diff = (apparentPower * apparentPower) - (realPower * realPower);
      reactivePower = (diff > 0.0f) ? sqrt(diff) : 0.0f; // Q = sqrt(S^2 - P^2)
    }

    bool gridAvailable = (vRms >= AC_MIN_VRMS);
    bool cap1 = false;
    bool cap2 = false;
    bool solar = false;
    bool wind = false;
    bool motor1 = false;
    bool motor2 = false;
    bool loadShed = false;

    // ── STEP 3: Control Logic (APFC & Load Demand Management)
    if (gridAvailable) {
      // ── GRID OPERATIONAL: Prioritize Grid, Lock Renewables OFF
      solar = false;
      wind = false;
      motor1 = true; // Critical Load ON
      motor2 = true; // Non-Critical Load ON (Plentiful grid capacity)
      loadShed = false;

      // Closed-Loop Automatic Power Factor Correction (Hysteresis)
      if (powerFactor < PF_THRESH_BOTH_ON) {
        cap1 = true;
        cap2 = true;
      } else if (powerFactor < PF_THRESH_ONE_ON) {
        cap1 = true;
        cap2 = false;
      } else if (powerFactor >= PF_THRESH_ALL_OFF) {
        cap1 = false;
        cap2 = false;
      }
    } else {
      // ── GRID FAILED: Island Microgrid Mode
      // Safely drop all capacitor banks (prevent resonance/damage on DC/inverter bus)
      cap1 = false;
      cap2 = false;
      solar = true; // Engage primary backup

      // Secondary backup activation heuristic
      if (iGen < GEN_MIN_CURRENT) {
        wind = true; // Engage wind if solar generation is insufficient
      }

      // Automated Load Demand Management (Dynamic Load Shedding)
      // Motor 1 is non-negotiable critical load; Motor 2 sheds during generation deficit
      motor1 = true;
      if (iGen < GEN_MIN_CURRENT) {
        motor2 = false;     // SHED LOAD 2
        loadShed = true;
      } else {
        motor2 = true;      // Generation sufficient, retain both loads
        loadShed = false;
      }
    }

    // ── STEP 4: Write Hardware Actuator Outputs
    setRelay(PIN_RELAY_CAP1, cap1);
    setRelay(PIN_RELAY_CAP2, cap2);
    setRelay(PIN_RELAY_SOLAR, solar);
    setRelay(PIN_RELAY_WIND, wind);
    setRelay(PIN_RELAY_MOTOR1, motor1);
    setRelay(PIN_RELAY_MOTOR2, motor2);

    // ── STEP 5: Cumulative Energy Accounting (kWh & Billing)
    unsigned long now = millis();
    float dtHours = (float)(now - lastEnergyTimestamp) / 3600000.0f;
    lastEnergyTimestamp = now;

    if (gridAvailable && realPower > 0.0f) {
      localEkWh += (realPower * dtHours) / 1000.0f;
    } else if (!gridAvailable && iGen > CURRENT_NOISE_FLOOR) {
      localESaved += (12.0f * iGen * dtHours) / 1000.0f; // 12V Microgrid Bus
    }
    float bill = localEkWh * TARIFF_PER_KWH;

    // ── STEP 6: Safely Update Inter-Core Shared State
    if (xSemaphoreTake(telemetryMutex, pdMS_TO_TICKS(40)) == pdTRUE) {
      sharedData.vrms           = vRms;
      sharedData.imains         = iMains;
      sharedData.iload          = iLoad;
      sharedData.igen           = iGen;
      sharedData.realPower      = realPower;
      sharedData.apparentPower  = apparentPower;
      sharedData.reactivePower  = reactivePower;
      sharedData.powerFactor    = powerFactor;
      sharedData.energyConsumed = localEkWh;
      sharedData.energySaved    = localESaved;
      sharedData.billAmount     = bill;
      sharedData.gridAvailable  = gridAvailable;
      sharedData.solarActive    = solar;
      sharedData.windActive     = wind;
      sharedData.cap1Active     = cap1;
      sharedData.cap2Active     = cap2;
      sharedData.motor1Active   = motor1;
      sharedData.motor2Active   = motor2;
      sharedData.loadShedActive = loadShed;
      xSemaphoreGive(telemetryMutex);
    }

    vTaskDelay(pdMS_TO_TICKS(150)); // Yield to prevent watchdog triggers
  }
}

// ================================================================================
// CORE 1: 5-PAGE ROTATING LCD DISPLAY & FIREBASE TELEMETRY CLOUD TASK
// ================================================================================
void displayCloudCore1Task(void *pvParameters) {
  unsigned long lastLcdFlip = 0;
  unsigned long lastCloudSync = 0;
  int lcdPage = 0;

  for (;;) {
    unsigned long currentMillis = millis();

    // ── UI Refresh Loop (Every 2.5 seconds)
    if (currentMillis - lastLcdFlip >= 2500) {
      lastLcdFlip = currentMillis;

      SystemTelemetry snapshot;
      if (xSemaphoreTake(telemetryMutex, pdMS_TO_TICKS(50)) == pdTRUE) {
        memcpy(&snapshot, (const void*)&sharedData, sizeof(SystemTelemetry));
        xSemaphoreGive(telemetryMutex);

        char line0[17], line1[17];
        lcd->clear();

        switch (lcdPage) {
          case 0: // Source & Grid Status
            snprintf(line0, 17, "SRC:%s %s %s",
                     snapshot.gridAvailable ? "GRID" : "----",
                     snapshot.solarActive ? "SOL" : "---",
                     snapshot.windActive  ? "WND" : "---");
            snprintf(line1, 17, "V:%3dV PF:%.2f", (int)snapshot.vrms, snapshot.powerFactor);
            break;

          case 1: // True Power Factor & Reactive Power
            snprintf(line0, 17, "PF:%.2f P:%4dW", snapshot.powerFactor, (int)snapshot.realPower);
            snprintf(line1, 17, "Q:%3dVAR CAP:%s", (int)snapshot.reactivePower,
                     (!snapshot.cap1Active && !snapshot.cap2Active) ? "OFF" :
                     (snapshot.cap1Active && !snapshot.cap2Active)  ? "10uF" : "20uF");
            break;

          case 2: // Load Currents & Automated Shedding Indicator
            snprintf(line0, 17, "ILd:%.1fA IGn:%.1fA", snapshot.iload, snapshot.igen);
            snprintf(line1, 17, "M1:%s M2:%s%s", 
                     snapshot.motor1Active ? "ON" : "OF",
                     snapshot.motor2Active ? "ON" : "OF",
                     snapshot.loadShedActive ? "(SHED)" : "");
            break;

          case 3: // Energy Consumption vs Renewable Savings
            snprintf(line0, 17, "Grid:%.3f kWh", snapshot.energyConsumed);
            snprintf(line1, 17, "Svd :%.3f kWh", snapshot.energySaved);
            break;

          case 4: // Energy Billing Estimate
            snprintf(line0, 17, "Bill:Rs.%7.2f", snapshot.billAmount);
            snprintf(line1, 17, "Tariff:8.50/unit");
            break;
        }

        lcd->setCursor(0, 0); lcd->print(line0);
        lcd->setCursor(0, 1); lcd->print(line1);
        lcdPage = (lcdPage + 1) % 5;
      }
    }

    // ── Cloud Telemetry Upload Loop (Every 6 seconds)
    if (currentMillis - lastCloudSync >= 6000 && WiFi.status() == WL_CONNECTED) {
      lastCloudSync = currentMillis;

      SystemTelemetry snap;
      if (xSemaphoreTake(telemetryMutex, pdMS_TO_TICKS(50)) == pdTRUE) {
        memcpy(&snap, (const void*)&sharedData, sizeof(SystemTelemetry));
        xSemaphoreGive(telemetryMutex);

        FirebaseJson liveJson;
        liveJson.set("timestamp",         getUptimeString());
        liveJson.set("grid_voltage_vrms", snap.vrms);
        liveJson.set("mains_current_a",   snap.imains);
        liveJson.set("load_current_a",    snap.iload);
        liveJson.set("generation_a",      snap.igen);
        liveJson.set("true_power_factor", snap.powerFactor);
        liveJson.set("real_power_w",      snap.realPower);
        liveJson.set("apparent_power_va", snap.apparentPower);
        liveJson.set("reactive_power_var",snap.reactivePower);
        liveJson.set("energy_kwh",        snap.energyConsumed);
        liveJson.set("energy_saved_kwh",  snap.energySaved);
        liveJson.set("bill_inr",          snap.billAmount);
        liveJson.set("cap_bank_uF",       (snap.cap1Active ? 10 : 0) + (snap.cap2Active ? 10 : 0));
        liveJson.set("load_shedding_on",  snap.loadShedActive);
        liveJson.set("source_active",     snap.gridAvailable ? "Grid 230V" : 
                                          (snap.solarActive && snap.windActive) ? "Solar+Wind" :
                                          snap.solarActive ? "Solar" : "Wind");

        Firebase.set(fbData, "/telemetry/live", liveJson);

        // Append historical log entry
        FirebaseJson logJson;
        logJson.set("ts",   getUptimeString());
        logJson.set("pf",   snap.powerFactor);
        logJson.set("p_w",  snap.realPower);
        logJson.set("v",    snap.vrms);
        logJson.set("shed", snap.loadShedActive);
        Firebase.push(fbData, "/telemetry/history", logJson);
      }
    }

    vTaskDelay(pdMS_TO_TICKS(100));
  }
}

// Scans I2C bus address (automatically distinguishes 0x27 from 0x3F)
uint8_t scanI2CAddress() {
  uint8_t selectedAddr = 0x27; // Standard default
  for (uint8_t addr = 0x08; addr <= 0x77; addr++) {
    Wire.beginTransmission(addr);
    if (Wire.endTransmission() == 0) {
      Serial.printf("[I2C] LCD backpack found at: 0x%02X\n", addr);
      selectedAddr = addr;
      break;
    }
  }
  return selectedAddr;
}

// Sets relay coil safely with state confirmation
void setRelay(int pin, bool state) {
  digitalWrite(pin, state ? RELAY_ON : RELAY_OFF);
}

// Returns formatted runtime string HH:MM:SS
String getUptimeString() {
  unsigned long totalSeconds = millis() / 1000;
  unsigned long seconds = totalSeconds % 60;
  unsigned long minutes = (totalSeconds / 60) % 60;
  unsigned long hours   = totalSeconds / 3600;
  char buf[12];
  snprintf(buf, sizeof(buf), "%02lu:%02lu:%02lu", hours, minutes, seconds);
  return String(buf);
}
